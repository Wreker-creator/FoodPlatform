docker compose down -v
docker compose build --no-cache
docker compose up -d
sleep 25
docker compose ps
docker compose logs kafka-init

# ============================================
# 1. CONFIRM CONSUMERS & PUBLISHERS
# ============================================
docker compose logs order-service | grep -iE "consumer starting|publisher starting"
docker compose logs inventory-service | grep -iE "consumer starting|publisher starting"
docker compose logs payment-service | grep -iE "consumer starting|publisher starting"
docker compose logs notification-service | grep -iE "consumer starting"

# ============================================
# 2. SEED INVENTORY
# ============================================
curl -X POST localhost:8081/inventory -H "Content-Type: application/json" -d '{"product_id":1,"quantity":100}'
docker exec -it inventory-db psql -U wreker -d inventory-db -c "SELECT * FROM inventory;"

# ============================================
# 3. SUCCESS PATH (Order 1: total = 19.98)
# ============================================
curl -X POST localhost:8080/orders -H "Content-Type: application/json" -d '{"customer_id":1,"items":[{"product_id":1,"quantity":2,"unit_price":9.99}]}'

# Allow outbox publishers (5-10s interval) and consumers to complete the saga
sleep 8

# Verify Outbox states across all 3 publishing services (all should be published = true)
docker exec -it order-db psql -U wreker -d order-db -c "SELECT id, event_type, published FROM outbox;"
docker exec -it inventory-db psql -U wreker -d inventory-db -c "SELECT id, event_type, published FROM outbox;"
docker exec -it payment-db psql -U wreker -d payment-db -c "SELECT id, event_type, published FROM outbox;"

# Verify final entity states
docker exec -it order-db psql -U wreker -d order-db -c "SELECT id, status FROM orders WHERE id = 1;"
docker exec -it inventory-db psql -U wreker -d inventory-db -c "SELECT * FROM inventory;"
docker exec -it payment-db psql -U wreker -d payment-db -c "SELECT * FROM payments WHERE order_id = 1;"
docker exec -it notification-db psql -U wreker -d notification-db -c "SELECT * FROM notifications WHERE order_id = 1;"
curl -s http://localhost:8084/notifications/1

# ============================================
# 4. PAYMENT-FAILURE PATH (Order 2: total = 1200)
# ============================================
curl -X POST localhost:8080/orders -H "Content-Type: application/json" -d '{"customer_id":1,"items":[{"product_id":1,"quantity":2,"unit_price":600}]}'

sleep 10

# Expected: order CANCELLED, stock restored to 98, payment FAILED, cancellation notification stored
docker exec -it order-db psql -U wreker -d order-db -c "SELECT id, status FROM orders WHERE id = 2;"
docker exec -it inventory-db psql -U wreker -d inventory-db -c "SELECT * FROM inventory;"
docker exec -it payment-db psql -U wreker -d payment-db -c "SELECT * FROM payments WHERE order_id = 2;"
docker exec -it notification-db psql -U wreker -d notification-db -c "SELECT * FROM notifications WHERE order_id = 2;"

# ============================================
# 5. INVENTORY-REJECTION PATH (Order 3: quantity = 9999)
# ============================================
curl -X POST localhost:8080/orders -H "Content-Type: application/json" -d '{"customer_id":1,"items":[{"product_id":1,"quantity":9999,"unit_price":9.99}]}'

sleep 10

# Expected: order CANCELLED, stock unchanged at 98, no payment row for order 3
docker exec -it order-db psql -U wreker -d order-db -c "SELECT id, status FROM orders WHERE id = 3;"
docker exec -it inventory-db psql -U wreker -d inventory-db -c "SELECT * FROM inventory;"
docker exec -it payment-db psql -U wreker -d payment-db -c "SELECT * FROM payments WHERE order_id = 3;"
docker exec -it notification-db psql -U wreker -d notification-db -c "SELECT * FROM notifications WHERE order_id = 3;"

# ============================================
# 6. ERROR SWEEP
# ============================================
docker compose logs order-service | grep -iE "error|fatal"
docker compose logs inventory-service | grep -iE "error|fatal"
docker compose logs payment-service | grep -iE "error|fatal"
docker compose logs notification-service | grep -iE "error|fatal"
