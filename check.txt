docker compose logs order-service | grep -iE "consumer starting|publisher starting"
docker compose logs inventory-service | grep -iE "consumer starting|publisher starting"
docker compose logs payment-service | grep -iE "consumer starting|publisher starting"
docker compose logs notification-service | grep -iE "consumer starting"

# Verify Prometheus Targets (All 4 jobs should report health: "up")
curl -s http://localhost:9090/api/v1/targets | grep -oE '"job":"[^"]+"|"health":"[^"]+"'

# Verify Grafana Provisioned Dashboard is live (Should return "Platform Health & Kafka Pipeline")
curl -s http://localhost:3000/api/search | grep -o '"title":"[^"]*"'


# ==============================================================================
# 2. SEED INITIAL INVENTORY
# ==============================================================================
curl -X POST localhost:8081/inventory -H "Content-Type: application/json" -d '{"product_id":1,"quantity":100}'
docker exec -i inventory-db psql -U wreker -d inventory-db -c "SELECT * FROM inventory;"


# ==============================================================================
# 3. PATH 1 (POSITIVE): HAPPY PATH (Order 1: 22 units @ 9.99 = 219.78 < 1000)
# Expected: Order CONFIRMED, Stock = 78 (100 - 22), Notification created
# ==============================================================================
curl -X POST localhost:9080/orders -H "Content-Type: application/json" -d '{"customer_id":1,"items":[{"product_id":1,"quantity":22,"unit_price":9.99}]}'

sleep 12

# Verify Outbox published = true & Idempotency (processed_events) across services
docker exec -i order-db psql -U wreker -d order-db -c "SELECT id, aggregate_key, event_type, published FROM outbox WHERE aggregate_key = '1';"
docker exec -i order-db psql -U wreker -d order-db -c "SELECT * FROM processed_events;"

docker exec -i inventory-db psql -U wreker -d inventory-db -c "SELECT id, aggregate_key, event_type, published FROM outbox WHERE aggregate_key = '1';"
docker exec -i inventory-db psql -U wreker -d inventory-db -c "SELECT * FROM processed_events;"

docker exec -i payment-db psql -U wreker -d payment-db -c "SELECT id, aggregate_key, event_type, published FROM outbox WHERE aggregate_key = '1';"
docker exec -i payment-db psql -U wreker -d payment-db -c "SELECT * FROM processed_events;"

# Verify final DB state transitions (Order 1 = CONFIRMED, Stock = 78)
docker exec -i order-db psql -U wreker -d order-db -c "SELECT id, customer_id, status, total_amount FROM orders WHERE id = 1;"
docker exec -i inventory-db psql -U wreker -d inventory-db -c "SELECT product_id, available_quantity FROM inventory WHERE product_id = 1;"
docker exec -i payment-db psql -U wreker -d payment-db -c "SELECT id, order_id, status, amount FROM payments WHERE order_id = 1;"
docker exec -i notification-db psql -U wreker -d notification-db -c "SELECT id, order_id, message FROM notifications WHERE order_id = 1;"


# ==============================================================================
# 4. PATH 2 (NEGATIVE): INSUFFICIENT INVENTORY (Order 2: 500 units > 78 available)
# Expected: InventoryRejected -> Order 2 CANCELLED, Stock remains 78
# ==============================================================================
curl -X POST localhost:9080/orders -H "Content-Type: application/json" -d '{"customer_id":1,"items":[{"product_id":1,"quantity":500,"unit_price":5.00}]}'

sleep 12

# Expect Order 2 status = CANCELLED
docker exec -i order-db psql -U wreker -d order-db -c "SELECT id, status, total_amount FROM orders WHERE id = 2;"
# Expect available_quantity is STILL 78
docker exec -i inventory-db psql -U wreker -d inventory-db -c "SELECT product_id, available_quantity FROM inventory WHERE product_id = 1;"
# Expect cancellation notification recorded for Order 2
docker exec -i notification-db psql -U wreker -d notification-db -c "SELECT id, order_id, message FROM notifications WHERE order_id = 2;"


# ==============================================================================
# 5. PATH 3 (NEGATIVE): PAYMENT FAILURE & SAGA COMPENSATION (Order 3: amount > 1000)
# Flow: Inventory reserves 10 units (78 -> 68) -> Payment fails (1500.00 > 1000)
#       -> Order 3 CANCELLED -> OrderCancelled (ReleaseInventory: true)
#       -> Inventory compensates & restores 10 units (68 -> 78)
# ==============================================================================
curl -X POST localhost:9080/orders -H "Content-Type: application/json" -d '{"customer_id":1,"items":[{"product_id":1,"quantity":10,"unit_price":150.00}]}'

sleep 15

# Expect Order 3 status = CANCELLED
docker exec -i order-db psql -U wreker -d order-db -c "SELECT id, status, total_amount FROM orders WHERE id = 3;"
# Expect Payment status = FAILED for order_id = 3
docker exec -i payment-db psql -U wreker -d payment-db -c "SELECT id, order_id, status, amount FROM payments WHERE order_id = 3;"
# Expect Stock restored back to 78 via compensating transaction!
docker exec -i inventory-db psql -U wreker -d inventory-db -c "SELECT product_id, available_quantity FROM inventory WHERE product_id = 1;"
# Expect OrderCancelled event in order-db outbox
docker exec -i order-db psql -U wreker -d order-db -c "SELECT id, aggregate_key, event_type, published FROM outbox WHERE aggregate_key = '3';"


# ==============================================================================
# 6. PATH 4 (NEGATIVE): IDEMPOTENCY & GUARDED STATE MACHINE CHAOS TEST
# Goal: Verify `processed_events`, `rowsAffected == 0`, and `status="duplicate"` metric
# ==============================================================================

# Step A: Inject duplicate "OrderCreated" event (event_id: "order-1") into order-events
docker exec -i kafka /opt/kafka/bin/kafka-console-producer.sh \
  --bootstrap-server localhost:9092 \
  --topic order-events \
  <<< '{"event_id":"order-1","event_type":"OrderCreated","payload":{"order_id":1,"customer_id":1,"items":[{"product_id":1,"quantity":22,"unit_price":9.99}]}}'

sleep 4

# Step B: Check Inventory logs & verify stock is STILL 78 (not deducted twice)
docker compose logs inventory-service | grep -i "idempotency"

# Step C: Inject stale out-of-order "InventoryReserved" event for already-CONFIRMED Order 1
docker exec -i kafka /opt/kafka/bin/kafka-console-producer.sh \
  --bootstrap-server localhost:9092 \
  --topic inventory-events \
  <<< '{"event_id":"inventory-999","event_type":"InventoryReserved","payload":{"order_id":1}}'

# Step D: Check Order Service logs for guarded update rejection

# Step E: Verify Prometheus recorded the `status="duplicate"` events

# ==============================================================================
# 7. PATH 5 (NEGATIVE): POISON PILL & DEAD LETTER QUEUE (DLQ) TEST
# Goal: Trigger unmarshal failure & 3-retry exhaustion -> verify DLQ topic & metrics
# ==============================================================================

# Case A: Completely malformed JSON envelope (immediate DLQ routing)
docker exec -i kafka /opt/kafka/bin/kafka-console-producer.sh \
  --bootstrap-server localhost:9092 \
  --topic inventory-events \
  <<< 'THIS_IS_CORRUPT_RAW_BYTES_NOT_JSON'

# Case B: Valid outer envelope, corrupt inner payload (triggers 3 retries + backoff -> DLQ)
docker exec -i kafka /opt/kafka/bin/kafka-console-producer.sh \
  --bootstrap-server localhost:9092 \
  --topic inventory-events \
  <<< '{"event_id":"poison-pill-1","event_type":"InventoryReserved","payload":"NOT_AN_OBJECT_CAUSES_RETRY_AND_DLQ"}'

sleep 5

# Verify retry warnings and DLQ routing in order-service logs
docker compose logs order-service | grep -iE "retrying message|routed poison pill to DLQ"

# Read the quarantined messages directly from the Kafka `dead-letter-events` topic
docker exec -i kafka /opt/kafka/bin/kafka-console-consumer.sh \
  --bootstrap-server localhost:9092 \
  --topic dead-letter-events \
  --from-beginning \
  --timeout-ms 5000

# Verify Prometheus `kafka_dlq_messages_total` counter incremented
curl -s 'http://localhost:9090/api/v1/query?query=kafka_dlq_messages_total'


# ==============================================================================
# 8. PATH 6 (CONCURRENCY): MULTI-REPLICA `FOR UPDATE SKIP LOCKED`
# Goal: Scale order-service to 3 replicas, fire 5 orders simultaneously
# Expected: Exactly 5 units deducted (78 -> 73), zero duplicate publishes
# ==============================================================================
docker compose up -d --scale order-service=3
sleep 10
docker compose ps

# Fire 5 concurrent orders (1 unit each)
for i in {1..5}; do
  curl -X POST localhost:9080/orders -H "Content-Type: application/json" -d '{"customer_id":1,"items":[{"product_id":1,"quantity":1,"unit_price":5.00}]}' &
done
wait

sleep 15

# Verify stock dropped by exactly 5 (78 - 5 = 73)
docker exec -i inventory-db psql -U wreker -d inventory-db -c "SELECT product_id, available_quantity FROM inventory WHERE product_id = 1;"

# Verify Prometheus DNS service discovery picked up all 3 order-service replica IPs
curl -s 'http://localhost:9090/api/v1/query?query=up\{job="order-service"\}'

# 9. PATH 7 (INFRASTRUCTURE CHAOS): KAFKA DOWN -> OUTBOX AGE & BACKLOG SPIKE
