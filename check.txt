# 5. PATH 5: THE IDEMPOTENCY & STATE MACHINE CHAOS TEST
# Target: Manually inject a duplicate event directly into Kafka
# Goal: Watch the new `processed_events` and `rowsAffected == 0` logic safely reject it
# ============================================

# Step A: Inject a duplicate "OrderCreated" event for Order 1 directly into the topic.
# (Adjust the JSON payload structure to match your exact EventEnvelope if needed)
docker exec -i kafka /opt/kafka/bin/kafka-console-producer.sh \
  --bootstrap-server localhost:9092 \
  --topic order-events \
  <<< '{"event_id":"order-1","event_type":"OrderCreated","payload":{"order_id":1,"customer_id":1,"items":[{"product_id":1,"quantity":2,"unit_price":9.99}]}}'

sleep 3

# Step B: Check Inventory Service Logs
# Expect: "idempotency check: event already processed, skipping"
docker compose logs inventory-service | grep -i "idempotency"

# Step C: Verify Inventory Math was NOT executed twice
# Expect: stock is exactly the same as it was at the end of Path 1 (e.g., 98, not 96)
docker exec -i inventory-db psql -U wreker -d inventory-db -c "SELECT product_id, available_quantity FROM inventory WHERE product_id = 1;"

# Step D: Inject a stale out-of-order event for Order 1
# Order 1 is already CONFIRMED. Let's send a delayed "InventoryReserved" event.
docker exec -i kafka /opt/kafka/bin/kafka-console-producer.sh \
  --bootstrap-server localhost:9092 \
  --topic inventory-events \
  <<< '{"event_id":"inventory-999","event_type":"InventoryReserved","payload":{"order_id":1}}'

sleep 3

# Step E: Check Order Service Logs
# Expect: "ignoring stale event: order not in expected state" (triggered by rowsAffected == 0)
